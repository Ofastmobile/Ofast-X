<?php
/**
 * Snippet Execution Engine
 * Three-tier security model
 */

class Ofast_Snippet_Engine {
    
    const TIMEOUT_LIMIT = 5; // seconds
    const MEMORY_LIMIT = '128M';
    
    /**
     * Execute snippets (called on init)
     */
    public function execute_active_snippets() {
        $snippets = $this->get_cached_active_snippets();
        
        foreach ($snippets as $snippet) {
            switch ($snippet->type) {
                case 'safe_action':
                    $this->execute_safe_action($snippet);
                    break;
                    
                case 'hook_based':
                    $this->execute_hook_based($snippet);
                    break;
                    
                case 'raw_code':
                    // REQUIRES APPROVAL
                    if ($snippet->status !== 'approved') {
                        continue;
                    }
                    $this->execute_raw_code_sandboxed($snippet);
                    break;
            }
        }
    }
    
    /**
     * Tier 1: Safe Actions (No code execution)
     */
    private function execute_safe_action($snippet) {
        $config = json_decode($snippet->action_config, true);
        
        switch ($snippet->action_type) {
            case 'add_header_code':
                // Only JS/CSS/HTML allowed
                if ($snippet->language === 'php') {
                    $this->log_security_violation($snippet->id, 'PHP in safe action');
                    return;
                }
                
                add_action('wp_head', function() use ($snippet) {
                    echo $snippet->code; // Already validated as non-PHP
                }, $config['priority'] ?? 10);
                break;
                
            case 'redirect_url':
                add_action('template_redirect', function() use ($config) {
                    if ($this->matches_conditions($config['conditions'])) {
                        wp_safe_redirect($config['target_url'], $config['code']);
                        exit;
                    }
                });
                break;
                
            case 'hide_admin_element':
                add_action('admin_head', function() use ($config) {
                    echo '<style>' . esc_html($config['css_selector']) . ' { display: none !important; }</style>';
                });
                break;
                
            // Add more safe actions...
        }
        
        $this->log_execution($snippet->id, 'safe_action');
    }
    
    /**
     * Tier 2: Hook-Based (WordPress hooks only)
     */
    private function execute_hook_based($snippet) {
        // Parse code to extract hook registration
        $hook = $snippet->hook_name; // e.g., 'init'
        $priority = $snippet->hook_priority ?? 10;
        
        // Build safe callable from code
        $callable = $this->build_safe_callable($snippet->code, $snippet->id);
        
        if ($callable) {
            add_action($hook, $callable, $priority);
            $this->log_execution($snippet->id, 'hook_based');
        }
    }
    
    /**
     * Build safe callable from code
     */
    private function build_safe_callable($code, $snippet_id) {
        // Validate first
        if (!$this->validate_hook_code($code)) {
            $this->log_security_violation($snippet_id, 'Invalid hook code');
            return null;
        }
        
        // Return closure with isolated scope
        return function() use ($code, $snippet_id) {
            // Set execution limits
            @set_time_limit(self::TIMEOUT_LIMIT);
            @ini_set('memory_limit', self::MEMORY_LIMIT);
            
            // Execute in try-catch
            try {
                // Use create_function for PHP 7.4 compatibility
                // OR anonymous function eval (still safer than raw eval)
                $func = create_function('', $code);
                
                if ($func === false) {
                    throw new Exception('Function creation failed');
                }
                
                $start = microtime(true);
                $result = $func();
                $duration = (microtime(true) - $start) * 1000;
                
                $this->log_execution($snippet_id, 'success', $duration);
                
                return $result;
                
            } catch (Throwable $e) {
                $this->log_error($snippet_id, $e->getMessage());
                $this->auto_deactivate($snippet_id);
            }
        };
    }
    
    /**
     * Tier 3: Raw Code (MAXIMUM SECURITY)
     */
    private function execute_raw_code_sandboxed($snippet) {
        // CRITICAL: Must be approved by admin
        if ($snippet->status !== 'approved') {
            $this->log_security_violation($snippet->id, 'Unapproved code execution attempt');
            return;
        }
        
        // Use subprocess isolation (WP-CLI if available)
        if ($this->has_wp_cli()) {
            $this->execute_via_cli($snippet);
        } else {
            // Fallback: Create temporary file and include with timeout
            $this->execute_via_tempfile($snippet);
        }
    }
    
    /**
     * Execute via WP-CLI (safest method)
     */
    private function execute_via_cli($snippet) {
        $temp_file = $this->create_temp_snippet_file($snippet);
        
        $cmd = sprintf(
            'timeout %d wp eval-file %s 2>&1',
            self::TIMEOUT_LIMIT,
            escapeshellarg($temp_file)
        );
        
        $output = shell_exec($cmd);
        
        unlink($temp_file);
        
        if (strpos($output, 'timeout') !== false) {
            $this->log_timeout($snippet->id);
            $this->auto_deactivate($snippet->id);
        } else {
            $this->log_execution($snippet->id, 'cli_success');
        }
    }
    
    /**
     * Validate hook-based code
     */
    private function validate_hook_code($code) {
        // 1. Syntax check
        if (!$this->php_syntax_valid($code)) {
            return false;
        }
        
        // 2. Forbidden functions (comprehensive list)
        $forbidden = [
            // Code execution
            'eval', 'assert', 'create_function', 'call_user_func',
            
            // System commands
            'exec', 'shell_exec', 'system', 'passthru', 'popen',
            'proc_open', 'pcntl_exec',
            
            // File operations
            'unlink', 'rmdir', 'rename', 'fwrite', 'file_put_contents',
            
            // Network
            'curl_exec', 'fsockopen', 'pfsockopen', 'socket_create',
            'stream_socket_client',
            
            // PHP config
            'dl', 'ini_set', 'ini_alter', 'apache_setenv', 'putenv',
            
            // Process control
            'exit', 'die', 'sleep', 'usleep', 'proc_*', 'posix_kill',
            
            // Obfuscation
            'base64_decode', 'gzinflate', 'gzuncompress', 'str_rot13',
            'hex2bin', 'convert_uudecode',
        ];
        
        foreach ($forbidden as $func) {
            if (preg_match('/\b' . preg_quote($func, '/') . '\s*\(/i', $code)) {
                return false;
            }
        }
        
        // 3. Detect obfuscation patterns
        $patterns = [
            '/\$\{["\']?\$/',           // Variable variables
            '/chr\s*\(\d+\)\s*\./',     // Character concatenation
            '/\\\x[0-9a-f]{2}/i',       // Hex escaping
            '/eval\s*\(.*base64/',      // Encoded eval

            [
                'status' => 'inactive', // Restore as inactive for safety

            ],
            ['id' => $snippet_id]
        );

    }

     * Permanently delete
     */
    public function delete_permanently($snippet_id) {
        global $wpdb;
        
        // Must be in trash first
        $snippet = $wpdb->get_row($wpdb->prepare(
            "SELECT * FROM {$wpdb->prefix}ofast_snippets WHERE id = %d AND status = 'trash'",
            $snippet_id

            return new WP_Error('not_in_trash', 'Snippet must be trashed first');

        // Save final revision
        $this->save_deletion_snapshot($snippet);
        
        // Delete
        $wpdb->delete($wpdb->prefix . 'ofast_snippets', ['id' => $snippet_id]);
        
        $this->log_action('permanently_deleted', $snippet_id);
    }

    public function cleanup_old_trash() {
