     * @since 1.0.0
     * @return void
     */
    public function init() {
        add_action('template_redirect', array($this, 'process'), 1);
        add_action('ofast_update_redirect_hit', array($this, 'process_hit_update'));
    }

    /**
     * Process redirects on frontend
     *
     * @since 1.0.0
     * @return void
     */
    public function process() {
        // Don't run in admin
        if (is_admin()) {
            return;
        }

        // Get current request
        $request_uri = isset($_SERVER['REQUEST_URI']) ? sanitize_text_field(wp_unslash($_SERVER['REQUEST_URI'])) : '';
        $request_path = wp_parse_url($request_uri, PHP_URL_PATH);

        if (empty($request_path)) {
            return;
        }

        // Get redirects from cache
        $redirects = $this->get_cached_redirects();

        if (empty($redirects)) {
            return;
        }

        // Find matching redirect
        foreach ($redirects as $redirect) {
            $match = $this->match_redirect($redirect, $request_path);

            if ($match) {
                $target = $match['target'];

                /**
                 * Filter redirect target before execution
                 *
                 * @since 1.0.0
                 * @param string $target      Target URL
                 * @param object $redirect    Redirect object
                 * @param string $request_uri Original request URI
                 */
                $target = apply_filters('ofast_redirect_target', $target, $redirect, $request_uri);

                /**
                 * Action before redirect execution
                 *
                 * @since 1.0.0
                 * @param object $redirect    Redirect object
                 * @param string $request_uri Original request URI
                 * @param string $target      Target URL
                 */
                do_action('ofast_before_redirect', $redirect, $request_uri, $target);

                // Schedule hit counter update (async)
                wp_schedule_single_event(time(), 'ofast_update_redirect_hit', array($redirect->id));

                // Execute redirect
                wp_safe_redirect($target, intval($redirect->type));
                exit;
            }
        }
    }

    /**
     * Match redirect against request path
     *
     * @since 1.0.0
     * @param object $redirect     Redirect object
     * @param string $request_path Request path
     * @return array|false Match data or false
     */
    private function match_redirect($redirect, $request_path) {
        if ($redirect->is_regex) {
            return $this->match_regex($redirect, $request_path);
        }

        return $this->match_exact($redirect, $request_path);
    }

    /**
     * Match exact redirect
     *
     * @since 1.0.0
     * @param object $redirect     Redirect object
     * @param string $request_path Request path
     * @return array|false Match data or false
     */
    private function match_exact($redirect, $request_path) {
        // Normalize paths (case-insensitive, trailing slashes)
        $source_normalized = rtrim(strtolower($redirect->source_url), '/');
        $request_normalized = rtrim(strtolower($request_path), '/');

        // Handle root path
        if ('' === $source_normalized) {
            $source_normalized = '/';
        }
        if ('' === $request_normalized) {
            $request_normalized = '/';
        }

        if ($request_normalized === $source_normalized) {
            return array(
                'matched' => true,
                'target' => $redirect->target_url,
            );
        }

        return false;
    }

    /**
     * Match regex redirect
     *
     * @since 1.0.0
     * @param object $redirect     Redirect object
     * @param string $request_path Request path
     * @return array|false Match data or false
     */
    private function match_regex($redirect, $request_path) {
        // Validate regex before execution
        if (!$this->is_valid_regex($redirect->source_url)) {
            return false;
        }

        $matches = array();

        // Use set_error_handler to catch regex errors
        set_error_handler(function () {
            // Suppress errors
        });

        $result = @preg_match('#' . $redirect->source_url . '#', $request_path, $matches);

        restore_error_handler();

        if ($result !== 1) {
            return false;
        }

        // Replace backreferences in target
        $target = $redirect->target_url;
        for ($i = 1; $i < count($matches); $i++) {
            $target = str_replace('$' . $i, $matches[$i], $target);
        }

        return array(
            'matched' => true,
            'target' => $target,
        );
    }

    /**
     * Validate regex pattern
     *
     * @since 1.0.0
     * @param string $pattern Regex pattern
     * @return bool True if valid, false otherwise
     */
    private function is_valid_regex($pattern) {
        // Length check
        if (strlen($pattern) > 200) {
            return false;
        }

        // Check for catastrophic backtracking patterns
        if (preg_match('/(\(.*\+.*\).*\+)|(\*.*\*)/', $pattern)) {
            return false;
        }

        // Test compilation
        set_error_handler(function () {
            // Suppress errors
        });

        $valid = @preg_match('#' . $pattern . '#', '') !== false;

        restore_error_handler();

        return $valid;
    }

    /**
     * Get redirects from cache
     *
     * @since 1.0.0
     * @return array Array of redirect objects
     */
    private function get_cached_redirects() {
        $cache_key = 'ofast_redirects_active';

        // Try object cache first (Redis/Memcached)
        $redirects = wp_cache_get($cache_key, 'ofast_redirects');

        if (false === $redirects) {
            // Try transient cache
            $redirects = get_transient($cache_key);

            if (false === $redirects) {
                // Cache miss - query database
                $redirects = $this->repository->find_active_redirects();

                // Store in transient
                set_transient($cache_key, $redirects, $this->cache_duration);
            }

            // Store in object cache
            wp_cache_set($cache_key, $redirects, 'ofast_redirects', $this->cache_duration);
        }

        return $redirects;
    }

    /**
     * Clear redirect cache
     *
     * @since 1.0.0
     * @return void
     */
    public function clear_cache() {
        delete_transient('ofast_redirects_active');
        wp_cache_delete('ofast_redirects_active', 'ofast_redirects');
    }

    /**
     * Process hit counter update (async worker)
     *
     * @since 1.0.0
     * @param int $id Redirect ID
     * @return void
     */
    public function process_hit_update($id) {
        $this->repository->increment_hits($id);
    }
}

<?php
/**
 * Ofast X - Redirects Importer
 *
 * Handles importing redirects from other plugins and JSON files.
 *
 * @package    Ofast_X
 * @subpackage Redirects
 * @since      1.0.0

/**
 * Redirects Importer Class
 *
 * @since 1.0.0
 */
class Ofast_Redirects_Importer {

    /**
     * Repository instance
     *
     * @since 1.0.0
     * @var Ofast_Redirects_Repository
     */
    private $repository;

    /**
     * Constructor
     *
     * @since 1.0.0
     * @param Ofast_Redirects_Repository $repository Repository instance
     */
    public function __construct($repository) {
        $this->repository = $repository;
    }

    /**
     * Detect available import sources
     *
     * @since 1.0.0
     * @return array Array of available sources
     */
    public function detect_sources() {
        global $wpdb;
        $sources = array();

        // Check Redirection plugin
        $redirection_table = $wpdb->prefix . 'redirection_items';
        if ($wpdb->get_var($wpdb->prepare("SHOW TABLES LIKE %s", $redirection_table)) === $redirection_table) {
            $count = $wpdb->get_var("SELECT COUNT(*) FROM {$redirection_table} WHERE action_type = 'url'");
            if ($count > 0) {
                $sources['redirection'] = array(
                    'name' => __('Redirection Plugin', 'ofast-x'),
                    'count' => $count,
                );
            }
        }

        // Check Safe Redirect Manager
        $srm_count = wp_count_posts('redirect_rule');
        if (isset($srm_count->publish) && $srm_count->publish > 0) {
            $sources['safe-redirect-manager'] = array(
                'name' => __('Safe Redirect Manager', 'ofast-x'),
                'count' => $srm_count->publish,
            );
        }

        // Check Simple 301 Redirects
        $simple_301 = get_option('301_redirects', array());
        if (!empty($simple_301) && is_array($simple_301)) {
            $sources['simple-301'] = array(
                'name' => __('Simple 301 Redirects', 'ofast-x'),
                'count' => count($simple_301),
            );
        }

        return $sources;
    }

    /**
     * Import from plugin
     *
     * @since 1.0.0
     * @param string $plugin Plugin identifier
     * @return array Result with count and message
     */
    public function import_from_plugin($plugin) {
        $imported = 0;

        switch ($plugin) {
            case 'redirection':
                $imported = $this->import_from_redirection();
                break;

            case 'safe-redirect-manager':
                $imported = $this->import_from_safe_redirect_manager();
                break;

            case 'simple-301':
                $imported = $this->import_from_simple_301();
                break;

            default:
                return array(
                    'success' => false,
                    'message' => __('Invalid import source.', 'ofast-x'),
                );
        }

        if ($imported > 0) {
            return array(
                'success' => true,
                'imported' => $imported,
                /* translators: %d: number of redirects imported */
                'message' => sprintf(_n('%d redirect imported as INACTIVE.', '%d redirects imported as INACTIVE.', $imported, 'ofast-x'), $imported),
            );
        }

        return array(
            'success' => true,
            'imported' => 0,
            'message' => __('No new redirects found to import.', 'ofast-x'),
        );
    }

    /**
     * Import from Redirection plugin
     *
     * @since 1.0.0
     * @return int Number of imported redirects
     */
    private function import_from_redirection() {
        global $wpdb;

        $source_table = $wpdb->prefix . 'redirection_items';
        $imported = 0;

        if ($wpdb->get_var($wpdb->prepare("SHOW TABLES LIKE %s", $source_table)) !== $source_table) {
            return 0;
        }

        $items = $wpdb->get_results("SELECT url, action_data, regex FROM {$source_table} WHERE action_type = 'url'");

        foreach ($items as $item) {
            if ($this->repository->source_exists($item->url)) {
                continue;
            }

            $this->repository->create(array(
                'source_url' => sanitize_text_field($item->url),
                'target_url' => esc_url_raw($item->action_data),
                'type' => '301',
                'is_regex' => $item->regex ? 1 : 0,
                'active' => 0, // Import as inactive
            ));

            $imported++;
        }

        return $imported;
    }

    /**
     * Import from Safe Redirect Manager
     *
     * @since 1.0.0
     * @return int Number of imported redirects
     */
    private function import_from_safe_redirect_manager() {
        $redirects = get_posts(array(
            'post_type' => 'redirect_rule',
            'posts_per_page' => -1,
            'post_status' => 'publish',
        ));

        $imported = 0;

        foreach ($redirects as $redirect) {
            $source = get_post_meta($redirect->ID, '_redirect_rule_from', true);
            $target = get_post_meta($redirect->ID, '_redirect_rule_to', true);
            $status = get_post_meta($redirect->ID, '_redirect_rule_status_code', true);

            if (empty($source) || empty($target)) {
                continue;
            }

            if ($this->repository->source_exists($source)) {
                continue;
            }

            $this->repository->create(array(
                'source_url' => sanitize_text_field($source),
                'target_url' => esc_url_raw($target),
                'type' => $status ? sanitize_text_field($status) : '301',
                'is_regex' => 0,
                'active' => 0,
            ));

            $imported++;
        }

        return $imported;
    }

    /**
     * Import from Simple 301 Redirects
     *
     * @since 1.0.0
     * @return int Number of imported redirects
     */
    private function import_from_simple_301() {
        $redirects = get_option('301_redirects', array());
        $imported = 0;

        if (!is_array($redirects)) {
            return 0;
        }

        foreach ($redirects as $source => $target) {
            if ($this->repository->source_exists($source)) {
                continue;
            }

            $this->repository->create(array(
                'source_url' => sanitize_text_field($source),
                'target_url' => esc_url_raw($target),
                'type' => '301',
                'is_regex' => 0,
                'active' => 0,
            ));

            $imported++;
        }

        return $imported;
    }

    /**
     * Import from JSON
     *
     * @since 1.0.0
     * @param string $json_data JSON data
     * @return array Result with count and message
     */
    public function import_from_json($json_data) {
        $data = json_decode($json_data, true);

        if (!$data || !isset($data['redirects']) || !is_array($data['redirects'])) {
            return array(
                'success' => false,
                'message' => __('Invalid JSON format.', 'ofast-x'),
            );
        }

        $imported = 0;

        foreach ($data['redirects'] as $redirect) {
            $source = isset($redirect['source_url']) ? sanitize_text_field($redirect['source_url']) : '';
            $target = isset($redirect['target_url']) ? esc_url_raw($redirect['target_url']) : '';

            if (empty($source) || empty($target)) {
                continue;
            }

            if ($this->repository->source_exists($source)) {
                continue;
            }

            $this->repository->create(array(
                'source_url' => $source,
                'target_url' => $target,
                'type' => isset($redirect['type']) ? sanitize_text_field($redirect['type']) : '301',
                'is_regex' => isset($redirect['is_regex']) ? intval($redirect['is_regex']) : 0,
                'active' => 0,
            ));

            $imported++;
        }

        if ($imported > 0) {
            return array(
                'success' => true,
                'imported' => $imported,
                /* translators: %d: number of redirects imported */
                'message' => sprintf(_n('%d redirect imported as INACTIVE.', '%d redirects imported as INACTIVE.', $imported, 'ofast-x'), $imported),
            );
        }

        return array(
            'success' => true,
            'imported' => 0,
            'message' => __('No new redirects found to import.', 'ofast-x'),
        );
    }

    /**
     * Export redirects to JSON
     *
     * @since 1.0.0
     * @param array $ids Optional array of redirect IDs to export
     * @return array Export data
     */
    public function export_to_json($ids = array()) {
        if (!empty($ids)) {
            $redirects = $this->repository->find_by_ids($ids);
        } else {
            $redirects = $this->repository->get_paginated(999999, 0); // Get all
        }

        return array(
            'plugin' => 'ofast-x',
            'version' => '1.0.0',
            'exported_at' => current_time('mysql'),
            'site_url' => get_site_url(),
            'redirects' => $redirects,
        );
    }
}

/**
 * Ofast X - Redirects Admin JavaScript
 *
 * @package    Ofast_X
 * @subpackage Redirects
 * @since      1.0.0
 */

(function($) {
    'use strict';

    var OfastRedirectsAdmin = {
        /**
         * Initialize
         */
        init: function() {
            this.bindEvents();
        },

        /**
         * Bind events
         */
        bindEvents: function() {
            $(document).on('click', '.ofast-redirect-toggle', this.toggleRedirect);
            $(document).on('click', '.ofast-redirect-delete', this.deleteRedirect);
            $(document).on('click', '.import-from-plugin', this.importFromPlugin);
            $(document).on('change', '#import-json-file', this.importFromJSON);
            $(document).on('click', '#export-redirects', this.exportRedirects);
            $(document).on('change', '#select-all-redirects', this.selectAll);
        },

        /**
         * Toggle redirect active/inactive
         */
        toggleRedirect: function(e) {
            e.preventDefault();
            
            var $btn = $(this);
            var id = $btn.data('id');
            var active = $btn.data('active');

            $btn.prop('disabled', true);

            $.post(ofastRedirects.ajaxUrl, {
                action: 'ofast_toggle_redirect',
                nonce: ofastRedirects.nonces.toggle,
                id: id,
                active: active
            }, function(response) {
                if (response.success) {
                    var newActive = response.data.active;
                    $btn.data('active', newActive);
                    $btn.text(newActive ? ofastRedirects.i18n.on : ofastRedirects.i18n.off);
                    $btn.toggleClass('button-primary', newActive === 1);
                }
            }).always(function() {
                $btn.prop('disabled', false);
            });
        },

        /**
         * Delete redirect
         */
        deleteRedirect: function(e) {
            e.preventDefault();
            
            if (!confirm(ofastRedirects.i18n.confirmDelete)) {
                return;
            }

            var $btn = $(this);
            var id = $btn.data('id');

            $btn.prop('disabled', true);

            $.post(ofastRedirects.ajaxUrl, {
                action: 'ofast_delete_redirect',
                nonce: ofastRedirects.nonces.delete,
                id: id
            }, function(response) {
                if (response.success) {
                    $btn.closest('tr').fadeOut(function() {
                        $(this).remove();
                        OfastRedirectsAdmin.checkEmptyTable();
                    });
                }
            }).fail(function() {
                $btn.prop('disabled', false);
            });
        },

        /**
         * Import from plugin
         */
        importFromPlugin: function(e) {
            e.preventDefault();
            
            var $btn = $(this);
            var plugin = $btn.data('plugin');
            var originalText = $btn.text();

            $btn.prop('disabled', true).text(ofastRedirects.i18n.importing);

            $.post(ofastRedirects.ajaxUrl, {
                action: 'ofast_import_redirects',
                nonce: ofastRedirects.nonces.import,
                plugin: plugin
            }, function(response) {
                if (response.success) {
                    alert(response.data.message);
                    if (response.data.imported > 0) {
                        location.reload();
                    }
                } else {
                    alert(response.data);
                }
            }).always(function() {
                $btn.prop('disabled', false).text(originalText);
            });
        },

        /**
         * Import from JSON file
         */
        importFromJSON: function(e) {
            var file = e.target.files[0];
            if (!file) {
                return;
            }

            var reader = new FileReader();
            reader.onload = function(e) {
                $.post(ofastRedirects.ajaxUrl, {
                    action: 'ofast_import_redirects',
                    nonce: ofastRedirects.nonces.import,
                    plugin: 'json',
                    json_data: e.target.result
                }, function(response) {
                    if (response.success) {
                        alert(response.data.message);
                        if (response.data.imported > 0) {
                            location.reload();
                        }
                    } else {
                        alert(response.data);
                    }
                });
            };
            reader.readAsText(file);
        },

        /**
         * Export redirects to JSON
         */
        exportRedirects: function(e) {
            e.preventDefault();
            
            var $btn = $(this);
            var originalText = $btn.text();

            $btn.prop('disabled', true).text(ofastRedirects.i18n.exporting);

            // Collect selected IDs
            var selectedIds = [];
            $('.redirect-checkbox:checked').each(function() {
                selectedIds.push($(this).val());
            });

            $.post(ofastRedirects.ajaxUrl, {
                action: 'ofast_export_redirects',
                nonce: ofastRedirects.nonces.export,
                ids: selectedIds
            }, function(response) {
                if (response.success) {
                    var blob = new Blob([JSON.stringify(response.data, null, 2)], {
                        type: 'application/json'
                    });
                    var url = URL.createObjectURL(blob);
                    var a = document.createElement('a');
                    a.href = url;
                    a.download = 'ofast-redirects-' + new Date().toISOString().split('T')[0] + '.json';
                    document.body.appendChild(a);
                    a.click();
                    document.body.removeChild(a);
                    URL.revokeObjectURL(url);
                }
            }).always(function() {
                $btn.prop('disabled', false).text(originalText);
            });
        },

        /**
         * Select all checkboxes
         */
        selectAll: function() {
            $('.redirect-checkbox').prop('checked', $(this).prop('checked'));
        },

        /**
         * Check if table is empty and show empty state
         */
        checkEmptyTable: function() {
            if ($('.wp-list-table tbody tr').length === 0) {
                location.reload();
            }
        }
    };

    // Initialize on document ready
    $(document).ready(function() {
        OfastRedirectsAdmin.init();
    });

})(jQuery);

/**
 * Redirects REST Controller Class
 *
 * @since 1.0.0
 */
class Ofast_Redirects_REST_Controller extends WP_REST_Controller {

    /**
     * Repository instance
     *
     * @since 1.0.0
     * @var Ofast_Redirects_Repository
     */
    private $repository;

    /**
     * Processor instance
     *
     * @since 1.0.0
     * @var Ofast_Redirects_Processor
     */
    private $processor;

    /**
     * Constructor
     *
     * @since 1.0.0
     * @param Ofast_Redirects_Repository $repository Repository instance
     * @param Ofast_Redirects_Processor  $processor  Processor instance
     */
    public function __construct($repository, $processor) {
        $this->repository = $repository;
        $this->processor = $processor;
        $this->namespace = 'ofast/v1';
        $this->rest_base = 'redirects';
    }

    /**
     * Register REST API routes
     *
     * @since 1.0.0
     * @return void
     */
    public function register_routes() {
        // List redirects
        register_rest_route($this->namespace, '/' . $this->rest_base, array(
            array(
                'methods' => WP_REST_Server::READABLE,
                'callback' => array($this, 'get_items'),
                'permission_callback' => array($this, 'get_items_permissions_check'),
                'args' => $this->get_collection_params(),
            ),
            array(
                'methods' => WP_REST_Server::CREATABLE,
                'callback' => array($this, 'create_item'),
                'permission_callback' => array($this, 'create_item_permissions_check'),
                'args' => $this->get_endpoint_args_for_item_schema(WP_REST_Server::CREATABLE),
            ),
        ));

        // Single redirect
        register_rest_route($this->namespace, '/' . $this->rest_base . '/(?P<id>[\d]+)', array(
            array(
                'methods' => WP_REST_Server::READABLE,
                'callback' => array($this, 'get_item'),
                'permission_callback' => array($this, 'get_item_permissions_check'),
            ),
            array(
                'methods' => WP_REST_Server::EDITABLE,
                'callback' => array($this, 'update_item'),
                'permission_callback' => array($this, 'update_item_permissions_check'),
                'args' => $this->get_endpoint_args_for_item_schema(WP_REST_Server::EDITABLE),
            ),
            array(
                'methods' => WP_REST_Server::DELETABLE,
                'callback' => array($this, 'delete_item'),
                'permission_callback' => array($this, 'delete_item_permissions_check'),
            ),
        ));
    }

    /**
     * Get collection of redirects
     *
     * @since 1.0.0
     * @param WP_REST_Request $request Full request object
     * @return WP_REST_Response|WP_Error
     */
    public function get_items($request) {
        $per_page = $request->get_param('per_page') ?: 20;
        $page = $request->get_param('page') ?: 1;
        $offset = ($page - 1) * $per_page;

        $redirects = $this->repository->get_paginated($per_page, $offset);
        $total = $this->repository->count();

        $data = array();
        foreach ($redirects as $redirect) {
            $data[] = $this->prepare_item_for_response($redirect, $request);
        }

        $response = rest_ensure_response($data);
        $response->header('X-WP-Total', $total);
        $response->header('X-WP-TotalPages', ceil($total / $per_page));

        return $response;
    }

    /**
     * Get single redirect
     *
     * @since 1.0.0
     * @param WP_REST_Request $request Full request object
     * @return WP_REST_Response|WP_Error
     */
    public function get_item($request) {
        $redirect = $this->repository->find_by_id($request->get_param('id'));

        if (!$redirect) {
            return new WP_Error('rest_redirect_not_found', __('Redirect not found.', 'ofast-x'), array('status' => 404));
        }

        return rest_ensure_response($this->prepare_item_for_response($redirect, $request));
    }

    /**
     * Create redirect
     *
     * @since 1.0.0
     * @param WP_REST_Request $request Full request object
     * @return WP_REST_Response|WP_Error
     */
    public function create_item($request) {
        $data = array(
            'source_url' => sanitize_text_field($request->get_param('source_url')),
            'target_url' => esc_url_raw($request->get_param('target_url')),
            'type' => sanitize_text_field($request->get_param('type')) ?: '301',
            'is_regex' => (int) $request->get_param('is_regex'),
            'active' => (int) $request->get_param('active'),
        );

        $id = $this->repository->create($data);

        if (!$id) {
            return new WP_Error('rest_redirect_create_failed', __('Failed to create redirect.', 'ofast-x'), array('status' => 500));
        }

        $this->processor->clear_cache();

        $redirect = $this->repository->find_by_id($id);

        return rest_ensure_response($this->prepare_item_for_response($redirect, $request));
    }

    /**
     * Update redirect
     *
     * @since 1.0.0
     * @param WP_REST_Request $request Full request object
     * @return WP_REST_Response|WP_Error
     */
    public function update_item($request) {
        $id = $request->get_param('id');
        $redirect = $this->repository->find_by_id($id);

        if (!$redirect) {
            return new WP_Error('rest_redirect_not_found', __('Redirect not found.', 'ofast-x'), array('status' => 404));
        }

        $data = array();
        
        if ($request->has_param('source_url')) {
            $data['source_url'] = sanitize_text_field($request->get_param('source_url'));
        }
        
        if ($request->has_param('target_url')) {
            $data['target_url'] = esc_url_raw($request->get_param('target_url'));
        }
        
        if ($request->has_param('type')) {
            $data['type'] = sanitize_text_field($request->get_param('type'));
        }
        
        if ($request->has_param('is_regex')) {
            $data['is_regex'] = (int) $request->get_param('is_regex');
        }
        
        if ($request->has_param('active')) {
            $data['active'] = (int) $request->get_param('active');
        }

        $result = $this->repository->update($id, $data);

        if (!$result) {
            return new WP_Error('rest_redirect_update_failed', __('Failed to update redirect.', 'ofast-x'), array('status' => 500));
        }

        $this->processor->clear_cache();

        $redirect = $this->repository->find_by_id($id);

        return rest_ensure_response($this->prepare_item_for_response($redirect, $request));
    }

    /**
     * Delete redirect
     *
     * @since 1.0.0
     * @param WP_REST_Request $request Full request object
     * @return WP_REST_Response|WP_Error
     */
    public function delete_item($request) {
        $id = $request->get_param('id');
        $redirect = $this->repository->find_by_id($id);

        if (!$redirect) {
            return new WP_Error('rest_redirect_not_found', __('Redirect not found.', 'ofast-x'), array('status' => 404));
        }

        $result = $this->repository->delete($id);

        if (!$result) {
            return new WP_Error('rest_redirect_delete_failed', __('Failed to delete redirect.', 'ofast-x'), array('status' => 500));
        }

        $this->processor->clear_cache();

        return rest_ensure_response(array('deleted' => true, 'previous' => $redirect));
    }

    /**
     * Prepare item for response
     *
     * @since 1.0.0
     * @param object          $item    Redirect object
     * @param WP_REST_Request $request Request object
     * @return array
     */
    public function prepare_item_for_response($item, $request) {
        return array(
            'id' => (int) $item->id,
            'source_url' => $item->source_url,
            'target_url' => $item->target_url,
            'type' => $item->type,
            'is_regex' => (bool) $item->is_regex,
            'active' => (bool) $item->active,
            'hits' => (int) $item->hits,
            'last_accessed' => $item->last_accessed,
            'created_at' => $item->created_at,
            'created_by' => (int) $item->created_by,
        );
    }

    /**
     * Check permissions for listing redirects
     *
     * @since 1.0.0
     * @param WP_REST_Request $request Request object
     * @return bool
     */
    public function get_items_permissions_check($request) {
        return current_user_can('manage_options');
    }

    /**
     * Check permissions for getting single redirect
     *
     * @since 1.0.0
     * @param WP_REST_Request $request Request object
     * @return bool
     */
    public function get_item_permissions_check($request) {
        return current_user_can('manage_options');
    }

    /**
     * Check permissions for creating redirect
     *
     * @since 1.0.0
     * @param WP_REST_Request $request Request object
     * @return bool
     */
    public function create_item_permissions_check($request) {
        return current_user_can('manage_options');
    }

    /**
     * Check permissions for updating redirect
     *
     * @since 1.0.0
     * @param WP_REST_Request $request Request object
     * @return bool
     */
    public function update_item_permissions_check($request) {
        return current_user_can('manage_options');
    }

    /**
     * Check permissions for deleting redirect
     *
     * @since 1.0.0
     * @param WP_REST_Request $request Request object
     * @return bool
     */
    public function delete_item_permissions_check($request) {
        return current_user_can('manage_options');
    }

    /**
     * Get collection parameters
     *
     * @since 1.0.0
     * @return array
     */
    public function get_collection_params() {
        return array(
            'page' => array(
                'description' => __('Current page of the collection.', 'ofast-x'),
                'type' => 'integer',
                'default' => 1,
                'minimum' => 1,
            ),
            'per_page' => array(
                'description' => __('Maximum number of items to be returned in result set.', 'ofast-x'),
                'type' => 'integer',
                'default' => 20,
                'minimum' => 1,
                'maximum' => 100,
            ),
        );
    }
}

/**
 * Main Redirects Module Class
 *
 * @since 1.0.0
 */
class Ofast_X_Redirects {

    /**
     * Repository instance
     *
     * @since 1.0.0
     * @var Ofast_Redirects_Repository
     */
    private $repository;

    /**
     * Processor instance
     *
     * @since 1.0.0
     * @var Ofast_Redirects_Processor
     */
    private $processor;

    /**
     * Importer instance
     *
     * @since 1.0.0
     * @var Ofast_Redirects_Importer
     */
    private $importer;

    /**
     * Admin instance
     *
     * @since 1.0.0
     * @var Ofast_Redirects_Admin
     */
    private $admin;

    /**
     * REST Controller instance
     *
     * @since 1.0.0
     * @var Ofast_Redirects_REST_Controller
     */
    private $rest_controller;

    /**
     * Initialize module
     *
     * @since 1.0.0
     * @return void
     */
    public function init() {
        // Load dependencies
        $this->load_dependencies();

        // Create database tables if needed
        if (Ofast_Redirects_Schema::needs_migration()) {
            Ofast_Redirects_Schema::migrate();
        }

        // Initialize components
        $this->repository = new Ofast_Redirects_Repository();
        $this->processor = new Ofast_Redirects_Processor($this->repository);
        $this->importer = new Ofast_Redirects_Importer($this->repository);

        // Initialize processor (handles frontend redirects)
        $this->processor->init();

        // Initialize admin interface
        if (is_admin()) {
            $this->admin = new Ofast_Redirects_Admin($this->repository, $this->importer, $this->processor);
            $this->admin->init();
        }

        // Initialize REST API
        $this->rest_controller = new Ofast_Redirects_REST_Controller($this->repository, $this->processor);
        add_action('rest_api_init', array($this->rest_controller, 'register_routes'));

        // Add WP-CLI support if available
        if (defined('WP_CLI') && WP_CLI) {
            $this->load_cli_commands();
        }
    }

    /**
     * Load dependencies
     *
     * @since 1.0.0
     * @return void
     */

        $base_path = OFAST_X_PLUGIN_DIR . 'modules/redirects/';

        require_once $base_path . 'class-ofast-redirects-importer.php';
        require_once $base_path . 'class-ofast-redirects-admin.php';
        require_once $base_path . 'class-ofast-redirects-rest-controller.php';
    }

    /**
     * Load WP-CLI commands
     *
     * @since 1.0.0
     * @return void
     */
    private function load_cli_commands() {

        WP_CLI::add_command('ofast redirects', 'Ofast_Redirects_CLI_Command');
    }

    /**
     * Get repository instance
     *
     * @since 1.0.0
     * @return Ofast_Redirects_Repository

    }

<?php
/**
 * Ofast X - Redirects WP-CLI Command
 *
 * Provides WP-CLI commands for managing redirects from the command line.

 * @package    Ofast_X
 * @subpackage Redirects
 * @since      1.0.0

// Exit if accessed directly

/**
 * Manage redirects via WP-CLI
 *
 * @since 1.0.0
 */
class Ofast_Redirects_CLI_Command {

    /**
     * Repository instance
     *
     * @since 1.0.0
     * @var Ofast_Redirects_Repository
     */
    private $repository;

    /**
     * Importer instance
     *
     * @since 1.0.0
     * @var Ofast_Redirects_Importer
     */
    private $importer;

    /**
     * Constructor
     *
     * @since 1.0.0
     */
    public function __construct() {
        $this->repository = new Ofast_Redirects_Repository();
        $this->importer = new Ofast_Redirects_Importer($this->repository);
    }

    /**
     * List all redirects
     *
     * ## OPTIONS
     *
     * [--active]
     * : Only show active redirects
     *
     * [--format=<format>]
     * : Render output in a particular format.
     * ---
     * default: table
     * options:
     *   - table
     *   - json
     *   - csv
     *   - yaml
     * ---
     *
     * ## EXAMPLES
     *
     *     wp ofast redirects list
     *     wp ofast redirects list --active
     *     wp ofast redirects list --format=json
     *
     * @since 1.0.0
     * @param array $args       Positional arguments
     * @param array $assoc_args Associative arguments
     * @return void
     */
    public function list($args, $assoc_args) {
        $redirects = $this->repository->get_paginated(999999, 0);

        if (isset($assoc_args['active'])) {
            $redirects = array_filter($redirects, function ($redirect) {
                return $redirect->active == 1;
            });
        }

        $format = isset($assoc_args['format']) ? $assoc_args['format'] : 'table';

        WP_CLI\Utils\format_items($format, $redirects, array(
            'id',
            'source_url',
            'target_url',
            'type',
            'is_regex',
            'active',
            'hits',
        ));
    }

    /**
     * Add a new redirect
     *
     * ## OPTIONS
     *
     * <source>
     * : Source URL path
     *
     * <target>
     * : Target URL
     *
     * [--type=<type>]
     * : Redirect type (301, 302, 307)
     * ---
     * default: 301
     * ---
     *
     * [--regex]
     * : Use regular expression matching
     *
     * [--inactive]
     * : Create redirect as inactive
     *
     * ## EXAMPLES
     *
     *     wp ofast redirects add /old-page /new-page
     *     wp ofast redirects add /old-page /new-page --type=302
     *     wp ofast redirects add '/blog/(.*)' '/news/$1' --regex
     *
     * @since 1.0.0
     * @param array $args       Positional arguments
     * @param array $assoc_args Associative arguments
     * @return void
     */
    public function add($args, $assoc_args) {
        list($source, $target) = $args;

        $data = array(
            'source_url' => sanitize_text_field($source),
            'target_url' => esc_url_raw($target),
            'type' => isset($assoc_args['type']) ? sanitize_text_field($assoc_args['type']) : '301',
            'is_regex' => isset($assoc_args['regex']) ? 1 : 0,
            'active' => isset($assoc_args['inactive']) ? 0 : 1,
        );

        $id = $this->repository->create($data);

        if ($id) {
            WP_CLI::success(sprintf('Redirect created with ID: %d', $id));
        } else {
            WP_CLI::error('Failed to create redirect.');
        }
    }

    /**
     * Delete a redirect
     *
     * ## OPTIONS
     *
     * <id>
     * : Redirect ID
     *
     * ## EXAMPLES
     *
     *     wp ofast redirects delete 123
     *
     * @since 1.0.0
     * @param array $args Positional arguments
     * @return void
     */
    public function delete($args) {
        list($id) = $args;

        $redirect = $this->repository->find_by_id($id);

        if (!$redirect) {
            WP_CLI::error('Redirect not found.');
        }

        $result = $this->repository->delete($id);

        if ($result) {
            WP_CLI::success(sprintf('Redirect #%d deleted.', $id));
        } else {
            WP_CLI::error('Failed to delete redirect.');
        }
    }

    /**
     * Activate a redirect
     *
     * ## OPTIONS
     *
     * <id>
     * : Redirect ID
     *
     * ## EXAMPLES
     *
     *     wp ofast redirects activate 123
     *
     * @since 1.0.0
     * @param array $args Positional arguments
     * @return void
     */
    public function activate($args) {
        list($id) = $args;

        $result = $this->repository->toggle_active($id, 1);

        if ($result) {
            WP_CLI::success(sprintf('Redirect #%d activated.', $id));
        } else {
            WP_CLI::error('Failed to activate redirect.');
        }
    }

    /**
     * Deactivate a redirect
     *
     * ## OPTIONS
     *
     * <id>
     * : Redirect ID
     *
     * ## EXAMPLES
     *
     *     wp ofast redirects deactivate 123
     *
     * @since 1.0.0
     * @param array $args Positional arguments
     * @return void
     */
    public function deactivate($args) {
        list($id) = $args;

        $result = $this->repository->toggle_active($id, 0);

        if ($result) {
            WP_CLI::success(sprintf('Redirect #%d deactivated.', $id));
        } else {
            WP_CLI::error('Failed to deactivate redirect.');
        }
    }

    /**
     * Import redirects from a JSON file
     *
     * ## OPTIONS
     *
     * <file>
     * : Path to JSON file
     *
     * ## EXAMPLES
     *
     *     wp ofast redirects import /path/to/redirects.json
     *
     * @since 1.0.0
     * @param array $args Positional arguments
     * @return void
     */
    public function import($args) {
        list($file) = $args;

        if (!file_exists($file)) {
            WP_CLI::error('File not found.');
        }

        $json_data = file_get_contents($file);
        $result = $this->importer->import_from_json($json_data);

        if ($result['success']) {
            WP_CLI::success($result['message']);
        } else {
            WP_CLI::error($result['message']);
        }
    }

    /**
     * Export redirects to a JSON file
     *
     * ## OPTIONS
     *
     * <file>
     * : Path to output JSON file
     *
     * ## EXAMPLES
     *
     *     wp ofast redirects export /path/to/redirects.json
     *
     * @since 1.0.0
     * @param array $args Positional arguments
     * @return void
     */
    public function export($args) {
        list($file) = $args;

        $data = $this->importer->export_to_json();
        $json = json_encode($data, JSON_PRETTY_PRINT);

        $result = file_put_contents($file, $json);

        if ($result) {
            WP_CLI::success(sprintf('Redirects exported to: %s', $file));
        } else {
            WP_CLI::error('Failed to write file.');
        }
    }

    /**
     * Clear the redirects cache
     *
     * ## EXAMPLES
     *
     *     wp ofast redirects clear-cache
     *
     * @since 1.0.0
     * @return void
     */
    public function clear_cache() {
        delete_transient('ofast_redirects_active');
        wp_cache_delete('ofast_redirects_active', 'ofast_redirects');

        WP_CLI::success('Redirects cache cleared.');
    }
}
