private function get_smtp_pulse() {
    global $wpdb;
    $table = $wpdb->prefix . 'ofast_smtp_log';
    
    if (!$this->table_exists('ofast_smtp_log')) {
        return array('status' => 'No Data', 'percent' => 100, 'total' => 0);
    }

    $total = $wpdb->get_var($wpdb->prepare(
        "SELECT COUNT(*) FROM {$table} WHERE sent_at >= DATE_SUB(NOW(), INTERVAL %d DAY)",
        30
    ));
    
    if ($total == 0) return array('status' => 'Ready', 'percent' => 100, 'total' => 0);

    $success = $wpdb->get_var($wpdb->prepare(
        "SELECT COUNT(*) FROM {$table} WHERE status = %s AND sent_at >= DATE_SUB(NOW(), INTERVAL %d DAY)",
        'success',
        30
    ));
    
    $percent = ($success / $total) * 100;

    return array(
        'status' => $percent > 95 ? 'Healthy' : 'Check Log',
        'percent' => round($percent, 2),
        'total' => (int) $total
    );
}

private function safe_get_data($method, $default = array()) {
    try {
        return $this->$method();
    } catch (Exception $e) {
        if (WP_DEBUG) {
            error_log("Ofast Dashboard: {$method} failed - " . $e->getMessage());
        }
        return $default;
    }
}

// AJAX handler for saving widget order
public function ajax_save_widget_order() {
    check_ajax_referer('ofast_dashboard_nonce', 'nonce');
    
    if (!current_user_can('manage_options')) {
        wp_send_json_error('Insufficient permissions');
    }
    
    $order = isset($_POST['order']) ? array_map('sanitize_text_field', $_POST['order']) : array();
    $allowed = array_keys($this->get_available_widgets());
    
    // Whitelist validation
    $order = array_filter($order, function($widget_id) use ($allowed) {
        return in_array(str_replace('#', '', $widget_id), $allowed);
    });
    
    update_user_meta(get_current_user_id(), 'ofast_widget_order', $order);
    wp_send_json_success();
}
